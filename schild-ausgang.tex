\SchildKopf[0.62]{Wenn du als Letzter das FabLab verlässt:}

\begin{Spalte}[0.78\linewidth]
\begin{itemize}
  \item Fenster schließen
  \item Tür zum Besprechungsraum schließen
  \item Geräte ausschalten, besonders: Ätzer, Laser inkl.\ Lüftung, 3D-Drucker, Lötkolben,
        Sicherungen der E-Werkstatt-Tische, Schneideplotter, Backofen, Temperofen
  \item Licht aus
  \item Eingangstür schließen (klemmt manchmal)
\end{itemize}
\end{Spalte}
